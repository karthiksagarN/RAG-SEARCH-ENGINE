# --- Core Web Framework ---
fastapi          # The main web framework used for building the API
uvicorn[standard]  # ASGI server to run the FastAPI application
pydantic         # Used by FastAPI for data validation and settings management (e.g., BaseModel)
python-multipart   # Required by FastAPI to handle file uploads (UploadFile)

# --- Environment & Utilities ---
python-dotenv    # To load environment variables from the .env file (os.getenv)

# --- LangChain Core & Models ---
langchain        # The core LangChain library (e.g., create_retrieval_chain)
langchain-core   # Core abstractions (e.g., ChatPromptTemplate)
langchain-openai # Specific integrations for OpenAI (ChatOpenAI, OpenAIEmbeddings)
openai           # The underlying client library used by langchain-openai

# --- LangChain Document Processing ---
langchain-community    # Contains document loaders (PyPDFLoader, TextLoader) and vector stores (Chroma)
langchain-text-splitters # Contains text splitters (RecursiveCharacterTextSplitter)
pypdf            # Required by PyPDFLoader to parse PDF files

# --- Vector Database ---
chromadb         # The client and database for the Chroma vector store
